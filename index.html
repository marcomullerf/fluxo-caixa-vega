// Cole a URL do seu Firebase Realtime Database aqui (sem barra no final): const FIREBASE_DB_URL = 'https://fluxo-53342-default-rtdb.firebaseio.com'; async function loadState() { try { // 1. Carrega primeiro do cache local para abrir instantaneamente const savedInitial = localStorage.getItem(STORAGE_KEY_INITIAL); const savedEntries = localStorage.getItem(STORAGE_KEY_ENTRIES); state.initialBalance = savedInitial !== null ? parseFloat(savedInitial) : DEFAULT_INITIAL_BALANCE; state.entries = savedEntries ? JSON.parse(savedEntries) : generateDefaultDataset(); render(); // 2. Busca a versão mais recente da nuvem (sincroniza entre dispositivos) const response = await fetch(`${FIREBASE_DB_URL}/vega_dados.json`); const cloudData = await response.json(); if (cloudData && Array.isArray(cloudData.entries) && cloudData.entries.length > 0) { state.initialBalance = cloudData.initialBalance ?? DEFAULT_INITIAL_BALANCE; state.entries = cloudData.entries; localStorage.setItem(STORAGE_KEY_INITIAL, String(state.initialBalance)); localStorage.setItem(STORAGE_KEY_ENTRIES, JSON.stringify(state.entries)); render(); } else { // Se o banco na nuvem ainda estiver vazio, envia os dados atuais para lá saveState(); } } catch (err) { console.warn('Modo offline ativo (usando dados locais):', err); } } async function saveState() { try { // Salva localmente no navegador localStorage.setItem(STORAGE_KEY_INITIAL, String(state.initialBalance)); localStorage.setItem(STORAGE_KEY_ENTRIES, JSON.stringify(state.entries)); // Envia para a nuvem para outros dispositivos acessarem await fetch(`${FIREBASE_DB_URL}/vega_dados.json`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ initialBalance: state.initialBalance, entries: state.entries, updatedAt: new Date().toISOString() }) }); } catch (err) { console.error('Erro ao sincronizar com a nuvem:', err); } }
